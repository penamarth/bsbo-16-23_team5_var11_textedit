```startuml
@startuml
autonumber
skinparam responseMessageBelowArrow true

actor User
participant "Editor" as Ctrl
participant "HighlightStrategyFactory" as Factory
participant "SearchStrategy" as Strategy
participant "Document" as Doc
participant "ParagraphComponent" as Para
participant "SentenceComponent" as Sent
participant "WordComponent" as Word

title Диаграмма последовательности: Поиск по файлу

User -> Ctrl: search_text("data")
activate Ctrl

Ctrl -> Factory: create_strategy("search", {query: "data"})
activate Factory

Factory -> Strategy: new SearchStrategy("data")
activate Strategy
Factory --> Ctrl: strategy (instance)
deactivate Factory

Ctrl -> Doc: apply_highlight(strategy)
activate Doc

Doc -> Para: apply_highlight(strategy)
activate Para

Para -> Sent: apply_highlight(strategy)
activate Sent

loop Для каждого слова в предложении
    Sent -> Word: apply_highlight(strategy)
    activate Word
    
    note right of Word: Word передает себя стратегии\nдля проверки
    Word -> Strategy: check_and_apply(self)
    activate Strategy
    
    alt Текст совпадает ("data" == "data")
        Strategy -> Word: set_search_highlight("orange")
        note left: Изменяется ТОЛЬКО слой фона.\nЦвет текста не трогаем.
    else Текст не совпадает
        Strategy -> Word: set_search_highlight(None)
        note left: Очистка слоя фона
    end
    
    Strategy --> Word: void
    deactivate Strategy
    
    Word --> Sent: void
    deactivate Word
end

Sent --> Para: void
deactivate Sent

Para --> Doc: void
deactivate Para

Doc --> Ctrl: void
deactivate Doc

Ctrl -> Doc: view_text()
activate Doc

loop При отрисовке каждого слова
    Doc -> Word: get_render_style()
    activate Word
    note right of Word: Слияние слоев:\nsyntax_color + search_highlight
    Word --> Doc: return (TextColor, BackgroundColor)
    deactivate Word
end

Doc --> User: Визуальный вывод (Текст + Фон)
deactivate Doc

deactivate Ctrl
@enduml
```

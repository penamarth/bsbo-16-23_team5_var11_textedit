# Диаграмма прецедентов UML для текстового редактора

```startuml
@startuml
left to right direction

' Настройка стилей для соответствия оригинальным цветам
skinparam actor {
    BackgroundColor #e1f5fe
    BorderColor #01579b
}

skinparam usecase {
    BackgroundColor #f3e5f5
    BorderColor #4a148c
}

actor "Пользователь" as actor1
actor "Принтер" as actor2

rectangle "Текстовый редактор" {
    usecase "Печать документа" as UC1
    usecase "Вставка текста" as UC2
    usecase "Сохранение документа в различных форматах" as UC3
    usecase "Подсветка синтаксиса" as UC4
    usecase "Поиск по файлу" as UC5
}

' Связи
actor1 --> UC1
actor1 --> UC2
actor1 --> UC3
actor1 --> UC4
actor1 --> UC5

UC1 --> actor2
@enduml
```

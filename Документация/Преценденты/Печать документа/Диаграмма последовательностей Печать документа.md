```startuml
@startuml
autonumber
skinparam responseMessageBelowArrow true

actor User
participant "Editor" as Ctrl
participant "PrintSettings" as Settings
participant "PrintManager" as PManager
participant "PrintPreview" as Preview
participant "Document" as Doc
participant "Printer" as Device

title Диаграмма последовательности: Печать документа

User -> Ctrl: set_print_settings(printer, copies, duplex, ...)
activate Ctrl

create Settings
Ctrl -> Settings: new PrintSettings(printer, copies, ...)
activate Settings
Settings --> Ctrl: settings (instance)
deactivate Settings

Ctrl --> User: Настройки сохранены
deactivate Ctrl

User -> Ctrl: show_preview()
activate Ctrl

Ctrl -> PManager: show_preview(document, settings)
activate PManager

PManager -> Doc: gettext()
activate Doc
Doc --> PManager: full_text
deactivate Doc

PManager -> Preview: generate_preview(full_text, settings)
activate Preview
Preview --> PManager: preview_image
deactivate Preview

PManager --> Ctrl: preview_image
deactivate PManager

Ctrl --> User: Отображение предпросмотра
deactivate Ctrl

User -> Ctrl: print_document()
activate Ctrl

Ctrl -> PManager: print(document, settings)
activate PManager

PManager -> Device: send_to_print(document, settings)
activate Device
Device --> PManager: job_id / status
deactivate Device

PManager --> Ctrl: Печать успешно запущена
deactivate PManager

Ctrl --> User: Уведомление о начале печати
deactivate Ctrl

@enduml
```

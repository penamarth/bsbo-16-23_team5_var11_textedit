@startuml
skinparam linetype ortho
top to bottom direction

package "Controller" {
  class Editor {
      -document: Document
      -export_fabric: ExportStrategyFabric
      -export_manager: ExportManager
      -highlight_factory: HighlightStrategyFactory
      -print_manager: PrintManager
      +new_document(name, path)
      +insert_text(text)
      +get_content()
      +search_text(query)
      +apply_syntax_highlight()
      +export(format_str)
      +print_document(printer_name, copies)
  }
}

package "Highlight" {
  class HighlightStrategyFactory {
      +create_strategy(type: String, params: Dict): IHighlightStrategy
  }

  interface IHighlightStrategy {
      +check_and_apply(word: WordComponent)
  }

  class SearchStrategy {
      -query: String
      +check_and_apply(word: WordComponent)
  }

  class SyntaxStrategy {
      -keywords: List
      -color: String
      +check_and_apply(word: WordComponent)
  }
}

package "Aggregate" {
  class Document {
      -name: String
      -path: String
      -content: List<ParagraphComponent>
      +gettext(): String
      +insert_text(text)
      +view_text()
      +get_components(): List<ParagraphComponent>
      +apply_highlight(strategy: IHighlightStrategy)
  }

  interface IComponent {
      +gettext()
      +apply_highlight(strategy: IHighlightStrategy)
  }

  class ParagraphComponent {
      -content: List<SentenceComponent>
      +add(text)
      +gettext(): String
      +get_components(): List<SentenceComponent>
      +apply_highlight(strategy: IHighlightStrategy)
  }

  class SentenceComponent {
      -content: List<WordComponent>
      +add(text)
      +gettext(): String
      +get_components(): List<WordComponent>
      +apply_highlight(strategy: IHighlightStrategy)
  }

  class WordComponent {
      -content: String
      -syntax_color: String
      -search_highlight: String
      +gettext(): String
      +set_syntax_color(color: String)
      +set_search_highlight(color: String)
      +get_render_style(): Tuple<String, String>
      +apply_highlight(strategy: IHighlightStrategy)
  }
}

package "Export" {
    class ExportManager {
        +export(strategy: IExportStrategy, doc: Document)
    }
    class ExportStrategyFabric {
        +create_strategy(format_str: String): IExportStrategy
    }
    class PdfExportStrategy {
        +export(doc: Document, path: String)
    }
    class JsonExportStrategy {
        +export(doc: Document, path: String)
    }
    interface IExportStrategy {
        +export(doc: Document, path: String)
    }
}

package "Print" {
    class PrintPreview {
        +generate_preview(doc: Document, settings: PrintSettings)
    }
    class PrintManager {
        +print(doc: Document, settings: PrintSettings)
        +show_preview(doc: Document, settings: PrintSettings)
    }
    class PrintSettings {
       -printer_name: String
       -copies: int
    }
}

' --- СВЯЗИ ---

' Editor relationships
Editor "1" *-- "1" Document
Editor --> PrintManager
Editor --> ExportManager
Editor *-- ExportStrategyFabric
Editor *-- HighlightStrategyFactory

' Highlight relationships
HighlightStrategyFactory ..> SearchStrategy : создает
HighlightStrategyFactory ..> SyntaxStrategy : создает
SearchStrategy ..|> IHighlightStrategy
SyntaxStrategy ..|> IHighlightStrategy

' Export relationships
ExportStrategyFabric ..> PdfExportStrategy : создаёт
ExportStrategyFabric ..> JsonExportStrategy : создаёт
ExportManager --> IExportStrategy : использует
PdfExportStrategy ..|> IExportStrategy
JsonExportStrategy ..|> IExportStrategy

' Document Composite structure
Document "1" *-- "*" ParagraphComponent
ParagraphComponent .up.|> IComponent
SentenceComponent .up.|> IComponent
WordComponent .up.|> IComponent

ParagraphComponent "1" *-- "*" SentenceComponent
SentenceComponent "1" *-- "*" WordComponent

' Document interactions
Document ..> IHighlightStrategy : принимает
IComponent ..> IHighlightStrategy : принимает
IHighlightStrategy ..> WordComponent : изменяет

' Print relationships
PrintManager "1" *-- "1" PrintPreview
PrintManager ..> PrintSettings : использует

@enduml
